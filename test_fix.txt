import os
import logging
from typing import Any, Dict, Optional

import google.auth
from google.auth.transport.requests import Request as GoogleAuthRequest
from flask import Flask, request, jsonify
from sqlalchemy import create_engine, text
from google.cloud.sql.connector import Connector, IPTypes

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Config
# ---------------------------------------------------------------------------
# "project:region:instance"
DB_INSTANCE_CONNECTION_NAME = os.getenv("DB_INSTANCE_CONNECTION_NAME")
DB_NAME = os.getenv("DB_NAME")

# Optional. If unset, derived from the runtime service account (recommended).
# For MySQL the IAM DB user is the SA email WITHOUT the ".gserviceaccount.com"
# suffix, e.g. "hold-api-sa@my-project.iam"
DB_IAM_USER = os.getenv("DB_IAM_USER")

# Set DB_PRIVATE_IP=true if the Cloud SQL instance has no public IP.
# Requires a Serverless VPC Access connector / Direct VPC egress on Cloud Run.
IP_TYPE = (
    IPTypes.PRIVATE
    if os.getenv("DB_PRIVATE_IP", "false").strip().lower() in ("1", "true", "yes")
    else IPTypes.PUBLIC
)

MAX_ERROR_LEN = int(os.getenv("MAX_ERROR_LEN", "2000"))

_connector: Optional[Connector] = None
_engine = None

app = Flask(__name__)


# ---------------------------------------------------------------------------
# IAM identity
# ---------------------------------------------------------------------------
def _detect_service_account_email() -> str:
    """Resolve the email of the identity this process is running as (ADC)."""
    creds, _ = google.auth.default(
        scopes=["https://www.googleapis.com/auth/cloud-platform"]
    )
    email = getattr(creds, "service_account_email", None)

    # On Cloud Run / GCE this is the literal string "default" until refreshed.
    if not email or email == "default":
        creds.refresh(GoogleAuthRequest())
        email = getattr(creds, "service_account_email", None)

    if not email or email == "default":
        raise RuntimeError(
            "Could not determine the runtime service account email. "
            "Set DB_IAM_USER explicitly."
        )
    return email


def get_iam_db_user() -> str:
    """MySQL IAM username = SA email minus the '.gserviceaccount.com' suffix."""
    if DB_IAM_USER:
        return DB_IAM_USER

    email = _detect_service_account_email()
    suffix = ".gserviceaccount.com"
    user = email[: -len(suffix)] if email.endswith(suffix) else email
    logger.info("Using IAM DB user: %s", user)
    return user


# ---------------------------------------------------------------------------
# Connection
# ---------------------------------------------------------------------------
def get_connector() -> Connector:
    global _connector
    if _connector is None:
        # No credentials= argument: Application Default Credentials are used,
        # i.e. the Cloud Run runtime service account.
        _connector = Connector(enable_iam_auth=True, ip_type=IP_TYPE)
    return _connector


def get_engine():
    global _engine
    if _engine is not None:
        return _engine

    if not all([DB_INSTANCE_CONNECTION_NAME, DB_NAME]):
        raise RuntimeError(
            "Missing DB env vars (DB_INSTANCE_CONNECTION_NAME/DB_NAME)"
        )

    iam_user = get_iam_db_user()

    def getconn():
        return get_connector().connect(
            DB_INSTANCE_CONNECTION_NAME,
            "pymysql",
            user=iam_user,
            db=DB_NAME,
            enable_iam_auth=True,   # no password: token is used instead
        )

    _engine = create_engine(
        "mysql+pymysql://",
        creator=getconn,
        pool_size=5,
        max_overflow=2,
        pool_timeout=30,
        # IAM access tokens live ~1h; recycle well before that.
        pool_recycle=1800,
        pool_pre_ping=True,
    )
    return _engine


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------
def account_last4(acct: str) -> str:
    acct = (acct or "").strip()
    return acct[-4:] if len(acct) >= 4 else acct


def truncate_error(msg: Any) -> str:
    s = str(msg or "")
    return s[:MAX_ERROR_LEN]


INSERT_HOLD_AUDIT = text(
    """
    INSERT INTO hold_audit (
        account_number_last4,
        account_number,
        hold_amount,
        hold_till_expired_date,
        hold_api_http_status,
        hold_api_result,
        hold_api_error
    ) VALUES (
        :account_number_last4,
        :account_number,
        :hold_amount,
        :hold_till_expired_date,
        :hold_api_http_status,
        :hold_api_result,
        :hold_api_error
    )
    """
)


def insert_hold_audit(payload: Dict[str, Any]) -> None:
    acct = payload.get("account_number")
    params = {
        "account_number_last4": account_last4(acct),
        "account_number": acct,
        "hold_amount": payload.get("hold_amount"),
        "hold_till_expired_date": payload.get("hold_till_expired_date"),
        "hold_api_http_status": payload.get("hold_api_http_status"),
        "hold_api_result": payload.get("hold_api_result"),
        "hold_api_error": truncate_error(payload.get("hold_api_error")),
    }
    engine = get_engine()
    with engine.begin() as conn:          # commits on success, rolls back on error
        conn.execute(INSERT_HOLD_AUDIT, params)


# ---------------------------------------------------------------------------
# Routes
# ---------------------------------------------------------------------------
@app.get("/healthz")
def healthz():
    """Liveness only - does not touch the DB."""
    return jsonify(status="ok"), 200


@app.get("/readyz")
def readyz():
    """Readiness - proves IAM auth and the DB round trip actually work."""
    try:
        with get_engine().connect() as conn:
            conn.execute(text("SELECT 1")).scalar_one()
        return jsonify(status="ok", db="reachable"), 200
    except Exception as exc:
        logger.exception("Readiness check failed")
        return jsonify(status="error", error=truncate_error(exc)), 503


@app.post("/hold-audit")
def hold_audit():
    payload = request.get_json(silent=True) or {}
    if not payload.get("account_number"):
        return jsonify(status="error", error="account_number is required"), 400

    try:
        insert_hold_audit(payload)
        return jsonify(status="ok"), 201
    except Exception as exc:
        logger.exception("Failed to write hold_audit row")
        return jsonify(status="error", error=truncate_error(exc)), 500


if __name__ == "__main__":
    # Cloud Run injects PORT. Use gunicorn in the container, not this.
    app.run(host="0.0.0.0", port=int(os.getenv("PORT", "8080")))
