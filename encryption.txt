"""
encryption.py — AES-256-CBC encryption for the final MHA response payload.

The key is a base64-encoded 32-byte value provided via config (Secret Manager on
Cloud Run). NEVER hard-code the key here.

Output format (what MHA expects):
    base64( iv[16 bytes] + ciphertext )   -> a single string.
The 16-byte IV is prepended to the ciphertext, then the whole thing is
base64-encoded. The receiver splits the first 16 bytes back off as the IV.
"""
from __future__ import annotations

import base64
import json
import os
from typing import Any

from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad

import config

IV_SIZE = 16  # AES block size (CBC)


def _key() -> bytes:
    """Return the raw 32-byte AES key, validating length. Raises if unset/bad."""
    b64 = config.MHA_RESPONSE_KEY_B64
    if not b64:
        raise RuntimeError(
            "MHA_RESPONSE_KEY_B64 is not set. Provide the base64 AES-256 key "
            "via env var / Secret Manager."
        )
    try:
        key = base64.b64decode(b64)
    except Exception as e:  # noqa: BLE001
        raise RuntimeError(f"MHA_RESPONSE_KEY_B64 is not valid base64: {e}") from e
    if len(key) != 32:
        raise RuntimeError(
            f"MHA_RESPONSE_KEY_B64 must decode to 32 bytes (AES-256); got {len(key)}."
        )
    return key


def encrypt_payload(payload: dict[str, Any]) -> str:
    """AES-256-CBC encrypt a JSON payload.

    Returns a single base64 string of (iv + ciphertext): the random 16-byte IV
    is prepended to the ciphertext before base64-encoding.
    """
    plain = json.dumps(payload, separators=(",", ":"), ensure_ascii=False)
    iv = os.urandom(IV_SIZE)
    cipher = AES.new(_key(), AES.MODE_CBC, iv)
    ciphertext = cipher.encrypt(pad(plain.encode("utf-8"), AES.block_size))
    return base64.b64encode(iv + ciphertext).decode("utf-8")


def decrypt_payload(base64_encrypted_text: str) -> dict[str, Any]:
    """Inverse of encrypt_payload — decrypt an MHA acknowledgement.

    Input is base64( iv[16] + ciphertext ); the first 16 bytes are the IV.
    Returns the decoded JSON dict. Raises on bad input / wrong key.
    """
    if not base64_encrypted_text:
        raise ValueError("Empty encrypted text")

    cleaned = base64_encrypted_text.strip().replace("\n", "").replace("\r", "").replace(" ", "")
    blob = base64.b64decode(cleaned)
    if len(blob) <= IV_SIZE:
        raise ValueError("Encrypted blob too short to contain IV + ciphertext")

    iv, ciphertext = blob[:IV_SIZE], blob[IV_SIZE:]
    cipher = AES.new(_key(), AES.MODE_CBC, iv)
    plain = unpad(cipher.decrypt(ciphertext), AES.block_size)
    return json.loads(plain.decode("utf-8"))


def is_enabled() -> bool:
    """True if an encryption key is configured (so callers can skip locally)."""
    return bool(config.MHA_RESPONSE_KEY_B64)
